const express = require('express'); const sqlite3 = require('sqlite3').verbose(); const bcrypt = require('bcryptjs'); const app = express(); const PORT = 3000; // Middleware app.use(express.json()); app.use(express.urlencoded({ extended: true })); // Initialize SQLite Database (creates medikiosk.db automatically) const db = new sqlite3.Database('./medikiosk.db', (err) => { if (err) console.error('Database opening error: ', err.message); else console.log('Connected to SQLite database.'); }); // Create tables for Patients, Doctors, and Triage Queues db.serialize(() => { db.run(`CREATE TABLE IF NOT EXISTS patients ( id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT, email TEXT UNIQUE, password TEXT )`); db.run(`CREATE TABLE IF NOT EXISTS doctors ( id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT, email TEXT UNIQUE, specialization TEXT, password TEXT )`); db.run(`CREATE TABLE IF NOT EXISTS triage_queue ( id INTEGER PRIMARY KEY AUTOINCREMENT, patient_id INTEGER, patient_name TEXT, symptoms TEXT, urgency_level TEXT, status TEXT DEFAULT 'Waiting', timestamp DATETIME DEFAULT CURRENT_TIMESTAMP )`); }); // ==================== API ROUTES ==================== // Patient Register app.post('/api/patient/register', async (req, res) => { const { name, email, password } = req.body; try { const hashedPassword = await bcrypt.hash(password, 10); db.run(`INSERT INTO patients (name, email, password) VALUES (?, ?, ?)`, [name, email, hashedPassword], function(err) { if (err) return res.status(400).json({ error: 'Email already exists or invalid data.' }); res.json({ success: true, patientId: this.lastID, name }); } ); } catch (e) { res.status(500).json({ error: 'Server error during registration.' }); } }); // Patient Login app.post('/api/patient/login', (req, res) => { const { email, password } = req.body; db.get(`SELECT * FROM patients WHERE email = ?`, [email], async (err, user) => { if (err || !user) return res.status(400).json({ error: 'Invalid email or password.' }); const isValid = await bcrypt.compare(password, user.password); if (!isValid) return res.status(400).json({ error: 'Invalid email or password.' }); res.json({ success: true, patientId: user.id, name: user.name }); }); }); // Doctor Register app.post('/api/doctor/register', async (req, res) => { const { name, email, specialization, password } = req.body; try { const hashedPassword = await bcrypt.hash(password, 10); db.run(`INSERT INTO doctors (name, email, specialization, password) VALUES (?, ?, ?, ?)`, [name, email, specialization, hashedPassword], function(err) { if (err) return res.status(400).json({ error: 'Email already exists.' }); res.json({ success: true, doctorId: this.lastID, name }); } ); } catch (e) { res.status(500).json({ error: 'Server error.' }); } }); // Doctor Login app.post('/api/doctor/login', (req, res) => { const { email, password } = req.body; db.get(`SELECT * FROM doctors WHERE email = ?`, [email], async (err, user) => { if (err || !user) return res.status(400).json({ error: 'Invalid email or password.' }); const isValid = await bcrypt.compare(password, user.password); if (!isValid) return res.status(400).json({ error: 'Invalid email or password.' }); res.json({ success: true, doctorId: user.id, name: user.name }); }); }); // Submit Triage Entry from Kiosk app.post('/api/triage/submit', (req, res) => { const { patientId, patientName, symptoms, urgencyLevel } = req.body; db.run(`INSERT INTO triage_queue (patient_id, patient_name, symptoms, urgency_level) VALUES (?, ?, ?, ?)`, [patientId, patientName, symptoms, urgencyLevel], function(err) { if (err) return res.status(500).json({ error: 'Failed to submit triage entry.' }); res.json({ success: true, queueId: this.lastID }); } ); }); // Fetch Live Queue for Doctors app.get('/api/triage/queue', (req, res) => { db.all(`SELECT * FROM triage_queue ORDER BY timestamp DESC`, [], (err, rows) => { if (err) return res.status(500).json({ error: 'Failed to fetch queue.' }); res.json(rows); }); }); // ==================== FRONTEND UI (Single Page Router) ==================== app.get('*', (req, res) => { res.send(` MediKiosk Hackathon App
`); }); app.listen(PORT, () => { console.log(`MediKiosk running locally at http://localhost:${PORT}`); });
